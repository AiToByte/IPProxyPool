    use std::sync::atomic::{AtomicUsize, Ordering};

    let sem = Arc::new(tokio::sync::Semaphore::new(2));
    let live = Arc::new(AtomicUsize::new(0));
    let high = Arc::new(AtomicUsize::new(0));
    let mut set = tokio::task::JoinSet::new();
    for _ in 0..8 {
        let (s, l, h) = (sem.clone(), live.clone(), high.clone());
        set.spawn(async move {
            let _permit = s.acquire_owned().await.expect("permit");
            let n = l.fetch_add(1, Ordering::SeqCst) + 1;
            h.fetch_max(n, Ordering::SeqCst);
            tokio::time::sleep(std::time::Duration::from_millis(20)).await;
            l.fetch_sub(1, Ordering::SeqCst);
        });
    }
    while set.join_next().await.is_some() {}
    assert!(high.load(Ordering::SeqCst) <= 2, "high={}", high.load(Ordering::SeqCst));
}
